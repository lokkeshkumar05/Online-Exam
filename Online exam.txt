let student = "";
let roll = "";
let timeLeft = 600;
let timerInterval;
let submitted = false;

function showPage(pageId) {

    document.querySelectorAll(".container section")
        .forEach(section => section.classList.add("hidden"));

    document.getElementById(pageId)
        .classList.remove("hidden");

}

document.getElementById("loginForm")
    .addEventListener("submit", function(e) {

        e.preventDefault();

        student = document.getElementById("studentName").value.trim();

        roll = document.getElementById("rollNumber").value.trim();

        if (!student || !roll) {
            alert("Please enter your details.");
            return;
        }

        showPage("instructionPage");

    });


function startExam() {

    showPage("examPage");

    document.getElementById("welcomeText").textContent =
        "Student: " + student + " | Roll No: " + roll;

    loadQuestions();

    timeLeft = 600;

    submitted = false;

    updateTimer();

    timerInterval = setInterval(function() {

        timeLeft--;

        updateTimer();

        if (timeLeft <= 0) {

            clearInterval(timerInterval);

            alert("Time is over! Your examination will be submitted.");

            calculateResult();

        }

    }, 1000);

}

// Convert HTML-sensitive characters into visible text so tags such as <a>, <img>,
// and comment syntax are displayed as quiz options instead of being parsed as HTML.
function escapeHTML(value) {
    return String(value).replace(/[&<>"']/g, function(character) {
        return {
            "&": "&amp;",
            "<": "&lt;",
            ">": "&gt;",
            '"': "&quot;",
            "'": "&#39;"
        }[character];
    });
}

function loadQuestions() {

    const container = document.getElementById("questions");

    container.innerHTML = "";

    quiz.forEach((q, index) => {

        let html = `
            <div class="question">
                <h3>${index + 1}. ${q.question}</h3>
        `;

        q.options.forEach((option, i) => {

            html += `
                <label class="option">
                    <input type="radio"
                        name="q${index}"
                        value="${i}"
                        onchange="updateProgress()">
                    ${escapeHTML(option)}
                </label>
            `;

        });

        html += `</div>`;

        container.innerHTML += html;

    });

    updateProgress();

}


function updateProgress() {

    let answered = 0;

    quiz.forEach((q, i) => {

        if (document.querySelector(
            'input[name="q' + i + '"]:checked'
        )) {

            answered++;

        }

    });

    document.getElementById("progressText").textContent =
        "Answered: " + answered + " / " + quiz.length;

    document.getElementById("progressBar").style.width =
        (answered / quiz.length * 100) + "%";

}

function updateTimer() {

    let minutes = Math.floor(timeLeft / 60);

    let seconds = timeLeft % 60;

    document.getElementById("timer").textContent =
        String(minutes).padStart(2, "0") + ":" +
        String(seconds).padStart(2, "0");

}


document.getElementById("examForm")
    .addEventListener("submit", function(e) {

        e.preventDefault();

        calculateResult();

    });


function calculateResult() {

    if (submitted) return;

    submitted = true;

    clearInterval(timerInterval);

    let correct = 0;

    let unanswered = 0;

    quiz.forEach((q, index) => {

        let selected = document.querySelector(
            'input[name="q' + index + '"]:checked'
        );

        if (!selected) {

            unanswered++;

        }

        else if (Number(selected.value) === q.answer) {

            correct++;

        }

    });

    let wrong = quiz.length - correct - unanswered;

    let percentage = (correct / quiz.length) * 100;


    document.getElementById("score").textContent =
        correct + "/" + quiz.length;

    document.getElementById("correct").textContent = correct;

    document.getElementById("wrong").textContent = wrong;

    document.getElementById("unanswered").textContent = unanswered;

    document.getElementById("percentage").textContent =
        percentage.toFixed(1);

    document.getElementById("resultStudent").textContent =
        "Student: " + student + " | Roll No: " + roll;

    document.getElementById("resultMessage").textContent =
        percentage >= 80 ? "Excellent Performance!" :
        percentage >= 50 ? "Good Job! Keep Practicing!" :
        "Keep Learning and Try Again!";

    showPage("resultPage");

}


function restartExam() {

    clearInterval(timerInterval);

    document.getElementById("loginForm").reset();

    showPage("loginPage");

}
